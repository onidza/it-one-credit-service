update config->

$body = (
  Get-Content "infra/kafka-connect/connector/credit-service-connector.json" -Raw |
  ConvertFrom-Json
).config | ConvertTo-Json -Depth 20

Invoke-RestMethod `
  -Method Put `
  -Uri "http://localhost:8083/connectors/credit-service-connector/config" `
  -ContentType "application/json" `
  -Body $body
